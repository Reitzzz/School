## 1. 贪心题下手顺序

1. 先找“每一步选什么最划算”。
2. 再确认选了之后不需要反悔。
3. 通常需要排序、堆、并查集等结构辅助。
4. 如果局部最优不一定推出整体最优，就不要硬套贪心，可能是 DP 或回溯。

## 2. 贪心：排序 / 优先队列 通用骨架

```cpp
// 1. 定义存储结构
struct Item {
    int property_A;
    int property_B;
};

// 2. 自定义贪心排序规则
bool cmp(const Item& a, const Item& b) {
    if (a.主关键字 != b.主关键字) {
        return a.主关键字 优于 b.主关键字;
    }
    return a.次关键字 优于 b.次关键字;
}

int main() {
    // 3. 读取数据
    vector<Item> items = 读取数据();

    // 4. 核心：按贪心策略排序
    sort(items.begin(), items.end(), cmp);

    // 5. 扫描并处理
    for (auto item : items) {
        if (当前 choice 合法 / 满足约束) {
            执行操作;
            更新状态;
        }
    }
}
```

## 3. Kruskal 最小生成树 通用骨架

```cpp
// 1. 定义边结构
struct Edge {
    int u, v;     // 端点
    int weight;   // 边权
};

// 2. 按边权从小到大排序
bool cmp(const Edge& a, const Edge& b) {
    return a.weight < b.weight;
}

// 3. 并查集（用于判断连通性，防止成环）
int find_set(int x) { /* 路径压缩查找 */ }
bool unite(int a, int b) { /* 按秩合并 */ }

int main() {
    vector<Edge> edges = 读取所有边();

    // 贪心策略：优先选权值最小的边
    sort(edges.begin(), edges.end(), cmp);

    初始化并查集;

    int total_weight = 0;
    int edge_count = 0;

    // 依次尝试加入每一条边
    for (auto edge : edges) {
        if (unite(edge.u, edge.v) == 成功合并) { // 两个端点原本不连通（不会成环）
            total_weight += edge.weight;
            edge_count++;
            
            // 优化：n个点的图，生成树只需要 n-1 条边
            if (edge_count == n - 1) { 
                break;
            }
        }
    }
}
```
