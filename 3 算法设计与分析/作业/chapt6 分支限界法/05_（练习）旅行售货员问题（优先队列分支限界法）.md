**题目描述**
某售货员要到若干城市去推销商品，已知各城市之间的路程（或旅费）。他要选定一条从驻地出发，经过每个城市一遍，最后回到驻地的路线，使总的路程（或总旅费）最短。这就是著名的旅行售货员问题（TSP）。
给定一个带权的完全有向图，请使用**优先队列式分支限界法**来求解旅行售货员问题的最短回路长度。

**输入格式**
第一行包含一个整数 $n$（$1 \le n \le 15$），表示城市的数量。城市编号从 1 到 $n$。
接下来 $n$ 行，每行包含 $n$ 个整数，表示城市之间的距离矩阵。第 $i$ 行第 $j$ 列的数字代表从城市 $i$ 到城市 $j$ 的距离。如果 $i = j$，距离用 -1 表示。

**输出格式**
输出一个整数，表示最短回路的总距离。

**输入样例**
```text
4
-1 30 6 4
30 -1 5 10
6 5 -1 20
4 10 20 -1
```

**输出样例**
```text
25
```
*(提示：最短路径为 1 -> 3 -> 2 -> 4 -> 1，代价为 6 + 5 + 10 + 4 = 25)*

---

# 解决思路

### 1. 核心需求：代价下界优先搜索
旅行售货员问题要求遍历所有城市并回到起点，其解空间树是一棵**排列树**。由于我们需要寻找最短路径，因此非常适合采用**优先队列式分支限界法**。我们在搜索过程中维护一个路径代价的**下界**（即这条路线走到底最少也要花多少代价），每次优先展开下界最小的结点。

### 2. 剪枝与限界函数
1. **下界估算策略**：
   - 最简单的估算方法是看“每离开一个城市，最少需要付出多少代价”。我们可以提前算出从每个城市出发的最短出边权重 `min_out[i]`。
   - 根结点的初始下界 `initial_lb` 就是所有城市的 `min_out[i]` 之和。
   - 当我们从城市 $u$ 走到城市 $v$ 时，我们已经确定了离开 $u$ 的代价是真实的边权 $w(u, v)$，而不再是原来的估算值 `min_out[u]`。因此，儿子结点的下界可以通过公式快速递推：`child.lb = parent.lb - min_out[u] + w(u, v)`。
2. **优先队列设计**：使用小根堆，按照结点的下界 `lb` 升序排列。
3. **剪枝条件**：
   - 维护一个全局变量 `best_cost`，记录目前找到的一条完整回路的最短长度。
   - 每次从优先队列弹出结点时，如果它的下界 `lb` 已经大于或等于 `best_cost`，说明沿着它继续往下走绝对不可能找到更优的解。因为堆顶是最小的下界，如果它都被淘汰了，队列里其余的结点更不可能，**直接结束搜索（break）**。

# 最终代码

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

const int INF = 1e9;

// 优先队列中的结点状态
struct Node {
    int level;          // 当前已经访问的城市数量（即树的层数，从 1 开始）
    int cc;             // 当前已经累积的真实路径代价
    int lb;             // 当前状态下，未来可能达到的最小代价下界
    vector<int> path;   // 记录当前的访问路径

    // 优先队列是小根堆，把下界最小的结点放在最前面优先扩展
    bool operator<(const Node& other) const {
        return lb > other.lb; 
    }
};

int main() {
    int n;
    if (!(cin >> n)) return 0;
    
    vector<vector<int>> graph(n + 1, vector<int>(n + 1));
    vector<int> min_out(n + 1, INF);
    
    // 读取距离矩阵，并预处理每个顶点的最小出边
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> graph[i][j];
            if (i != j && graph[i][j] != -1) {
                min_out[i] = min(min_out[i], graph[i][j]);
            }
        }
    }
    
    // 计算根节点的初始下界
    int initial_lb = 0;
    for (int i = 1; i <= n; i++) {
        if (min_out[i] != INF) {
            initial_lb += min_out[i];
        }
    }
    
    priority_queue<Node> pq;
    int best_cost = INF;
    
    // 初始化根结点：从城市 1 出发，此时 path 只有 1 个城市
    Node root = {1, 0, initial_lb, {1}};
    pq.push(root);
    
    // 优先队列式分支限界核心
    while (!pq.empty()) {
        Node curr = pq.top();
        pq.pop();
        
        // 剪枝：如果当前结点的下界已经不小于已知的最优回路，则直接终止搜索
        if (curr.lb >= best_cost) {
            break; 
        }
        
        // 找到了一条到达倒数第二层的路径（即准备走最后一个城市并回起点）
        if (curr.level == n - 1) {
            // 找出唯一没有被访问的那个城市
            int last_city = -1;
            vector<bool> visited(n + 1, false);
            for (int city : curr.path) visited[city] = true;
            for (int i = 1; i <= n; i++) {
                if (!visited[i]) {
                    last_city = i;
                    break;
                }
            }
            
            // 检查倒数第二个城市到最后一个城市，以及最后一个城市回起点的路是否连通
            int u = curr.path.back();
            if (graph[u][last_city] != -1 && graph[last_city][1] != -1) {
                int total_cost = curr.cc + graph[u][last_city] + graph[last_city][1];
                if (total_cost < best_cost) {
                    best_cost = total_cost;
                }
            }
            continue;
        }
        
        // 扩展产生子结点
        int u = curr.path.back();
        vector<bool> visited(n + 1, false);
        for (int city : curr.path) visited[city] = true;
        
        for (int v = 1; v <= n; v++) {
            if (!visited[v] && graph[u][v] != -1) {
                Node child;
                child.level = curr.level + 1;
                child.cc = curr.cc + graph[u][v];
                // 子结点的下界：父结点下界 - 去掉原来的理想化估算 + 换成实际的花费
                child.lb = curr.lb - min_out[u] + graph[u][v]; 
                
                child.path = curr.path;
                child.path.push_back(v);
                
                // 只有下界小于最优解时，才加入队列
                if (child.lb < best_cost) {
                    pq.push(child);
                }
            }
        }
    }
    
    cout << best_cost << "\n";
    return 0;
}
```
